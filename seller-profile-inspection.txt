import "server-only";

import bcrypt from "bcryptjs";
import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";
import { ObjectId } from "mongodb";
import { getDatabase } from "@/lib/mongodb";

export const ACCOUNT_ROLES = [
  "farmer",
  "buyer",
  "supplier",
  "worker",
  "expert",
] as const;

export type AccountRole = (typeof ACCOUNT_ROLES)[number];

export type AuthUser = {
  id: string;
  name: string;
  email: string;
  phone: string;
  role: AccountRole;
};

const COOKIE_NAME = "agrivanta_session";
const SESSION_DURATION_SECONDS = 60 * 60 * 24 * 7;

function getSecret() {
  const secret = process.env.AUTH_SECRET;

  if (!secret || secret.length < 32) {
    throw new Error(
      "AUTH_SECRET must contain a strong secret of at least 32 characters."
    );
  }

  return new TextEncoder().encode(secret);
}

export async function hashPassword(password: string) {
  return bcrypt.hash(password, 12);
}

export async function verifyPassword(
  password: string,
  passwordHash: string
) {
  return bcrypt.compare(password, passwordHash);
}

export async function createSession(user: AuthUser) {
  const token = await new SignJWT({
    role: user.role,
    name: user.name,
    email: user.email,
  })
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(user.id)
    .setIssuedAt()
    .setExpirationTime(`${SESSION_DURATION_SECONDS}s`)
    .sign(getSecret());

  const cookieStore = await cookies();

  cookieStore.set(COOKIE_NAME, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: SESSION_DURATION_SECONDS,
  });
}

export async function destroySession() {
  const cookieStore = await cookies();

  cookieStore.set(COOKIE_NAME, "", {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    path: "/",
    maxAge: 0,
  });
}

export async function getSessionClaims() {
  const cookieStore = await cookies();
  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (!token) {
    return null;
  }

  try {
    const { payload } = await jwtVerify(token, getSecret(), {
      algorithms: ["HS256"],
    });

    if (
      typeof payload.sub !== "string" ||
      typeof payload.role !== "string" ||
      !ACCOUNT_ROLES.includes(payload.role as AccountRole)
    ) {
      return null;
    }

    return {
      userId: payload.sub,
      role: payload.role as AccountRole,
    };
  } catch {
    return null;
  }
}

export async function getCurrentUser(): Promise<AuthUser | null> {
  const claims = await getSessionClaims();

  if (!claims || !ObjectId.isValid(claims.userId)) {
    return null;
  }

  const db = await getDatabase();

  const user = await db.collection("users").findOne(
    { _id: new ObjectId(claims.userId) },
    {
      projection: {
        name: 1,
        email: 1,
        phone: 1,
        role: 1,
      },
    }
  );

  if (!user || !ACCOUNT_ROLES.includes(user.role as AccountRole)) {
    return null;
  }

  return {
    id: user._id.toString(),
    name: String(user.name ?? ""),
    email: String(user.email ?? ""),
    phone: String(user.phone ?? ""),
    role: user.role as AccountRole,
  };
}

import { NextResponse } from "next/server";
import { MongoServerError } from "mongodb";
import { getDatabase } from "@/lib/mongodb";
import {
  ACCOUNT_ROLES,
  createSession,
  hashPassword,
  type AccountRole,
} from "@/lib/auth";



const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export async function POST(request: Request) {
  try {
    const body: unknown = await request.json();

    if (!body || typeof body !== "object" || Array.isArray(body)) {
      return NextResponse.json(
        { success: false, message: "Invalid request data." },
        { status: 400 }
      );
    }

    const data = body as Record<string, unknown>;

    const name =
      typeof data.name === "string" ? data.name.trim() : "";
    const email =
      typeof data.email === "string"
        ? data.email.trim().toLowerCase()
        : "";
    const phone =
      typeof data.phone === "string" ? data.phone.trim() : "";
    const password =
      typeof data.password === "string" ? data.password : "";
    const role =
      typeof data.role === "string" ? data.role : "";

    if (name.length < 2 || name.length > 80) {
      return NextResponse.json(
        { success: false, message: "Name must be 2â€“80 characters." },
        { status: 400 }
      );
    }

    if (email.length > 254 || !emailPattern.test(email)) {
      return NextResponse.json(
        { success: false, message: "Enter a valid email address." },
        { status: 400 }
      );
    }

    if (phone.length > 30) {
      return NextResponse.json(
        { success: false, message: "Phone number is too long." },
        { status: 400 }
      );
    }

    if (password.length < 10 || password.length > 72) {
      return NextResponse.json(
        {
          success: false,
          message: "Password must contain 10â€“72 characters.",
        },
        { status: 400 }
      );
    }

    if (!ACCOUNT_ROLES.includes(role as AccountRole)) {
      return NextResponse.json(
        { success: false, message: "Select a valid account type." },
        { status: 400 }
      );
    }

    const db = await getDatabase();
    const users = db.collection("users");

    const existingUser = await users.findOne(
      { email },
      { projection: { _id: 1 } }
    );

    if (existingUser) {
      return NextResponse.json(
        {
          success: false,
          message: "An account with this email already exists.",
        },
        { status: 409 }
      );
    }

    const passwordHash = await hashPassword(password);
    const now = new Date();

    const result = await users.insertOne({
      name,
      email,
      phone,
      role,
      passwordHash,
      createdAt: now,
      updatedAt: now,
    });

    const user = {
      id: result.insertedId.toString(),
      name,
      email,
      phone,
      role: role as AccountRole,
    };

    await createSession(user);

    return NextResponse.json(
      {
        success: true,
        message: "Your AgriVanta account has been created.",
        user,
      },
      { status: 201 }
    );
  } catch (error) {
    if (
      error instanceof MongoServerError &&
      error.code === 11000
    ) {
      return NextResponse.json(
        {
          success: false,
          message: "An account with this email already exists.",
        },
        { status: 409 }
      );
    }

    console.error("Registration error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "Registration failed. Please try again.",
      },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { getDatabase } from "@/lib/mongodb";
import { getCurrentUser } from "@/lib/auth";

const allowedTypes = [
  "product",
  "hire",
  "service",
  "worker",
  "expert",
] as const;

type ListingType = (typeof allowedTypes)[number];

const rolePermissions: Record<string, readonly ListingType[]> = {
  farmer: ["product", "hire"],
  buyer: [],
  supplier: ["product", "service"],
  worker: ["worker"],
  expert: ["expert", "service"],
};

const fieldLimits: Record<string, number> = {
  title: 120,
  category: 80,
  subcategory: 100,
  description: 5000,
  priceUnit: 40,
  location: 120,
  image: 2000,
  experience: 500,
  availability: 500,
};

const MAX_IMAGES = 8;
const MAX_VIDEOS = 2;
const MAX_MEDIA = 8;

function validText(
  value: unknown,
  maxLength: number
): value is string {
  return (
    typeof value === "string" &&
    value.trim().length > 0 &&
    value.trim().length <= maxLength
  );
}

function isCloudinaryUrl(value: unknown): value is string {
  if (typeof value !== "string" || value.length > 2000) {
    return false;
  }

  try {
    const url = new URL(value);
    return (
      url.protocol === "https:" &&
      url.hostname === "res.cloudinary.com"
    );
  } catch {
    return false;
  }
}

function isValidSpecifications(value: unknown): boolean {
  if (
    typeof value !== "object" ||
    value === null ||
    Array.isArray(value)
  ) {
    return false;
  }

  const entries = Object.entries(value);

  if (entries.length > 40) {
    return false;
  }

  return entries.every(([key, item]) => {
    if (!key.trim() || key.length > 100) return false;

    if (typeof item === "string") {
      return item.length <= 500;
    }

    if (typeof item === "number") {
      return Number.isFinite(item) && item >= 0;
    }

    return typeof item === "boolean";
  });
}

function isValidMediaArray(
  value: unknown,
  maxLength: number
): value is string[] {
  return (
    Array.isArray(value) &&
    value.length <= maxLength &&
    value.every(isCloudinaryUrl)
  );
}

function isValidMediaObjects(value: unknown): boolean {
  if (!Array.isArray(value) || value.length > MAX_MEDIA) {
    return false;
  }

  return value.every((item) => {
    if (
      typeof item !== "object" ||
      item === null ||
      Array.isArray(item)
    ) {
      return false;
    }

    const media = item as Record<string, unknown>;

    return (
      isCloudinaryUrl(media.secure_url) &&
      (media.type === "image" || media.type === "video") &&
      (media.resource_type === "image" ||
        media.resource_type === "video")
    );
  });
}

export async function GET(request: NextRequest) {
  try {
    const db = await getDatabase();
    const params = request.nextUrl.searchParams;

    const type = params.get("type");
    const category = params.get("category");
    const subcategory = params.get("subcategory");
    const location = params.get("location");
    const search = params.get("q")?.trim();

    if (
      type &&
      !(allowedTypes as readonly string[]).includes(type)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid listing type." },
        { status: 400 }
      );
    }

    if (
      (category && category.length > 80) ||
      (subcategory && subcategory.length > 100) ||
      (location && location.length > 120) ||
      (search && search.length > 100)
    ) {
      return NextResponse.json(
        { success: false, message: "Search filters are too long." },
        { status: 400 }
      );
    }

    const filter: Record<string, unknown> = {
      status: "active",
    };

    if (type) filter.type = type;
    if (category) filter.category = category;
    if (subcategory) filter.subcategory = subcategory;
    if (location) filter.location = location;

    if (search) {
      const escaped = search.replace(
        /[.*+?^${}()|[\]\\]/g,
        "\\$&"
      );

      filter.$or = [
        { title: { $regex: escaped, $options: "i" } },
        { category: { $regex: escaped, $options: "i" } },
        { subcategory: { $regex: escaped, $options: "i" } },
        { description: { $regex: escaped, $options: "i" } },
        { location: { $regex: escaped, $options: "i" } },
        { skills: { $regex: escaped, $options: "i" } },
      ];
    }

    const listings = await db
      .collection("listings")
      .find(filter)
      .sort({ createdAt: -1 })
      .limit(100)
      .toArray();

    return NextResponse.json({
      success: true,
      count: listings.length,
      listings: listings.map((document) => {
        const {
          _id,
          userId: _userId,
          userEmail: _userEmail,
          sellerId: _sellerId,
          sellerEmail: _sellerEmail,
          ownerId: _ownerId,
          ownerEmail: _ownerEmail,
          createdBy: _createdBy,
          ...publicListing
        } = document;

        return {
          ...publicListing,
          id: _id.toString(),
        };
      }),
    });
  } catch (error) {
    console.error("Failed to retrieve listings:", error);

    return NextResponse.json(
      { success: false, message: "Unable to load listings." },
      { status: 500 }
    );
  }
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        {
          success: false,
          message: "Please log in before creating a listing.",
        },
        { status: 401 }
      );
    }

    const permittedTypes = rolePermissions[user.role] ?? [];

    if (permittedTypes.length === 0) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Your account type does not have permission to create listings.",
        },
        { status: 403 }
      );
    }

    const body: unknown = await request.json();

    if (
      !body ||
      typeof body !== "object" ||
      Array.isArray(body)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid request body." },
        { status: 400 }
      );
    }

    const data = body as Record<string, unknown>;

    const requiredLimits = {
      title: fieldLimits.title,
      category: fieldLimits.category,
      description: fieldLimits.description,
      priceUnit: fieldLimits.priceUnit,
      location: fieldLimits.location,
    };

    const requiredFieldsValid = Object.entries(requiredLimits).every(
      ([key, maxLength]) => validText(data[key], maxLength)
    );

    if (
      !requiredFieldsValid ||
      typeof data.type !== "string" ||
      !(allowedTypes as readonly string[]).includes(data.type) ||
      !permittedTypes.includes(data.type as ListingType) ||
      typeof data.price !== "number" ||
      !Number.isFinite(data.price) ||
      data.price < 0 ||
      data.price > 1_000_000_000
    ) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Please provide valid listing details for your account type.",
        },
        { status: 400 }
      );
    }

    const subcategory = data.subcategory;

    if (
      subcategory !== undefined &&
      !validText(subcategory, fieldLimits.subcategory)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid listing subcategory." },
        { status: 400 }
      );
    }

    if (
      data.specifications !== undefined &&
      !isValidSpecifications(data.specifications)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid listing specifications." },
        { status: 400 }
      );
    }

    if (
      data.image !== undefined &&
      data.image !== "" &&
      !isCloudinaryUrl(data.image)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid listing image URL." },
        { status: 400 }
      );
    }

    const images =
      data.images === undefined
        ? []
        : data.images;

    const videos =
      data.videos === undefined
        ? []
        : data.videos;

    if (!isValidMediaArray(images, MAX_IMAGES)) {
      return NextResponse.json(
        { success: false, message: "Invalid listing images." },
        { status: 400 }
      );
    }

    if (!isValidMediaArray(videos, MAX_VIDEOS)) {
      return NextResponse.json(
        { success: false, message: "Invalid listing videos." },
        { status: 400 }
      );
    }

    if (
      data.media !== undefined &&
      !isValidMediaObjects(data.media)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid listing media." },
        { status: 400 }
      );
    }

    if (images.length + videos.length > MAX_MEDIA) {
      return NextResponse.json(
        {
          success: false,
          message: `You can attach up to ${MAX_MEDIA} media files.`,
        },
        { status: 400 }
      );
    }

    if (
      data.skills !== undefined &&
      (!Array.isArray(data.skills) ||
        data.skills.length > 50 ||
        !data.skills.every(
          (skill) =>
            typeof skill === "string" &&
            skill.trim().length > 0 &&
            skill.trim().length <= 80
        ))
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid skills." },
        { status: 400 }
      );
    }

    if (
      data.experience !== undefined &&
      (typeof data.experience !== "string" ||
        data.experience.length > fieldLimits.experience)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid experience." },
        { status: 400 }
      );
    }

    if (
      data.availability !== undefined &&
      (typeof data.availability !== "string" ||
        data.availability.length > fieldLimits.availability)
    ) {
      return NextResponse.json(
        { success: false, message: "Invalid availability." },
        { status: 400 }
      );
    }

    const primaryImage =
      typeof data.image === "string" && data.image.trim()
        ? data.image.trim()
        : images[0] ?? "";

    if (!primaryImage) {
      return NextResponse.json(
        {
          success: false,
          message: "Upload at least one image for your listing.",
        },
        { status: 400 }
      );
    }

    const db = await getDatabase();
    const now = new Date();

    const listing = {
      title: (data.title as string).trim(),
      type: data.type as ListingType,
      category: (data.category as string).trim(),
      subcategory:
        typeof subcategory === "string" ? subcategory.trim() : "",
      specifications:
        data.specifications &&
        typeof data.specifications === "object" &&
        !Array.isArray(data.specifications)
          ? data.specifications
          : {},
      description: (data.description as string).trim(),
      price: data.price,
      priceUnit: (data.priceUnit as string).trim(),
      location: (data.location as string).trim(),

      // Identity is derived from the authenticated user, not the request.
      seller: user.name,
      userId: user.id,
      userEmail: user.email,
      sellerId: user.id,
      sellerEmail: user.email,
      ownerId: user.id,
      ownerEmail: user.email,
      createdBy: user.id,
      sellerRole: user.role,

      image: primaryImage,
      images,
      videos,
      media: Array.isArray(data.media) ? data.media : [],
      skills: Array.isArray(data.skills)
        ? data.skills.map((skill) => (skill as string).trim())
        : [],
      experience:
        typeof data.experience === "string"
          ? data.experience.trim()
          : "",
      availability:
        typeof data.availability === "string"
          ? data.availability.trim()
          : "",
      mediaCount: Math.max(
        images.length + videos.length,
        primaryImage ? 1 : 0
      ),
      status: "active",
      createdAt: now,
      updatedAt: now,
    };

    const result = await db.collection("listings").insertOne(listing);

    return NextResponse.json(
      {
        success: true,
        message: "Listing created successfully.",
        id: result.insertedId.toString(),
      },
      { status: 201 }
    );
  } catch (error) {
    if (error instanceof SyntaxError) {
      return NextResponse.json(
        { success: false, message: "Invalid JSON request body." },
        { status: 400 }
      );
    }

    console.error("Failed to create listing:", error);

    return NextResponse.json(
      { success: false, message: "Unable to create listing." },
      { status: 500 }
    );
  }
}

import { connection } from "next/server";

export async function GET() {
  await connection();

  try {
    const user = await getCurrentUser();

    // Keep the rest of your existing GET logic here.

    if (!user) {
      return NextResponse.json(
        { success: false, message: "Please log in to view your listings." },
        { status: 401 }
      );
    }

    const ownerConditions: Record<string, string>[] = [];

    if (user.id) {
      ownerConditions.push(
        { userId: user.id },
        { sellerId: user.id },
        { ownerId: user.id },
        { createdBy: user.id }
      );
    }

    if (user.email) {
      ownerConditions.push(
        { userEmail: user.email },
        { sellerEmail: user.email },

      );
    }

    if (ownerConditions.length === 0) {
      return NextResponse.json({
        success: true,
        count: 0,
        listings: [],
      });
    }

    const db = await getDatabase();

    const documents = await db
      .collection("listings")
      .find({ $or: ownerConditions })
      .sort({ createdAt: -1 })
      .limit(100)
      .toArray();

    const listings = documents.map((document) => {
      const { _id, ...listing } = document;

      return {
        ...listing,
        id: _id.toString(),
      };
    });

    return NextResponse.json({
      success: true,
      count: listings.length,
      listings,
    });
  } catch (error) {
    console.error("Failed to load user's listings:", error);

    return NextResponse.json(
      { success: false, message: "Unable to load your listings." },
      { status: 500 }
    );
  }
}

function getListingOwnerId(listing: Record<string, unknown>) {

  for (const field of fields) {
    const value = listing[field];
    if (typeof value === "string" && value.trim()) {
      return value.trim();
    }
  }

  return null;
}

export async function POST(request: NextRequest) {
  try {
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        { error: "Please log in to contact a seller." },
        { status: 401 }
      );
    }

    const body = await request.json();
    const listingId =
      typeof body.listingId === "string" ? body.listingId.trim() : "";
    const message =
      typeof body.message === "string" ? body.message.trim() : "";

    if (!ObjectId.isValid(listingId)) {
      return NextResponse.json(
        { error: "A valid listing ID is required." },
        { status: 400 }
      );
    }

    if (message.length < 10 || message.length > 3000) {
      return NextResponse.json(
        { error: "Your message must be between 10 and 3000 characters." },
        { status: 400 }
      );
    }

    const db = await getDatabase();

    const listing = await db.collection("listings").findOne({
      _id: new ObjectId(listingId),
      status: "active",
    });

    if (!listing) {
      return NextResponse.json(

    );

    if (!sellerId) {

        { status: 409 }
      );
    }

    if (sellerId === user.id) {
      return NextResponse.json(
        { error: "You cannot send an inquiry about your own listing." },
        { status: 400 }
      );
    }

    const now = new Date();

    const inquiry = {
      listingId,
      listingTitle: String(listing.title ?? "Untitled listing"),
      listingType: String(listing.type ?? "product"),

      sellerId,
      sellerName: String(
        listing.seller ?? listing.sellerName ?? "Seller"
      ),
      sellerEmail:
        typeof listing.sellerEmail === "string"
          ? listing.sellerEmail
          : typeof listing.userEmail === "string"
            ? listing.userEmail
            : "",

      buyerId: user.id,
      buyerName: user.name,
      buyerEmail: user.email,
      buyerPhone: user.phone,

      message,
      status: "new",
      createdAt: now,
      updatedAt: now,
    };

    const result = await db.collection("inquiries").insertOne(inquiry);

    return NextResponse.json(
      {
        message: "Your inquiry has been sent successfully.",
        inquiry: {
          id: result.insertedId.toString(),
          listingId: inquiry.listingId,
          listingTitle: inquiry.listingTitle,
          status: inquiry.status,
          createdAt: inquiry.createdAt,
        },
      },

    );
  } catch (error) {
    console.error("Create inquiry error:", error);

    return NextResponse.json(
      { error: "Unable to send your inquiry right now." },
      { status: 500 }
    );
  }
}

  async function submitInquiry(event: FormEvent<HTMLFormElement>) {

    event.preventDefault();

    if (!contactListing || inquirySending) return;

    const message = inquiryMessage.trim();

    if (message.length < 10) {

      setInquiryFeedback("Please enter at least 10 characters.");

      setInquiryFeedbackType("error");

      return;

    }

    setInquirySending(true);

    setInquiryFeedback("");

    setInquiryFeedbackType("");

    try {

      const response = await fetch("/api/inquiries", {

        method: "POST",

        headers: { "Content-Type": "application/json" },

        credentials: "include",

        body: JSON.stringify({

          listingId: contactListing.id,

          message,

        }),

      });

      const data = await response.json().catch(() => ({}));

      if (response.status === 401) {

        setInquiryFeedback("Please log in before contacting a seller.");

        setInquiryFeedbackType("error");

        return;

      }

      if (!response.ok) {

        throw new Error(data.error || "Unable to send your inquiry.");

      }

      setInquiryFeedback(

        data.message || "Your inquiry has been sent successfully."

      );

      setInquiryFeedbackType("success");

      setInquiryMessage("");

    } catch (error) {

      setInquiryFeedback(

        error instanceof Error

          ? error.message

          : "Something went wrong while sending your inquiry."

      );

      setInquiryFeedbackType("error");

    } finally {

      setInquirySending(false);

    }

  }

                  inquiryFeedbackType === "success"

                    ? "bg-[#EAF0E6] text-[#17472D]"

                    : "bg-red-50 text-red-700"

                }`}

              >

                {inquiryFeedback}

                {inquiryFeedbackType === "error" &&

                  inquiryFeedback.toLowerCase().includes("log in") && (

                    <>

                      {" "}

                      <a href="/login" className="font-bold underline">

                        Log in

                      </a>

                    </>

                  )}

              </div>

            )}

            {inquiryFeedbackType !== "success" && (

              <form onSubmit={submitInquiry} className="mt-4 space-y-4">

                <label

                  htmlFor="inquiry-message"

                  className="block text-sm font-semibold text-[#263D2D]"

                >

                  Your message

                </label>

                <textarea

                  id="inquiry-message"

                  value={inquiryMessage}

                  onChange={(event) => setInquiryMessage(event.target.value)}

                  required

                  minLength={10}

                  maxLength={3000}

                  rows={5}

                  placeholder="Ask about availability, price, delivery, or other details..."

                  className="w-full resize-y rounded-xl border border-[#D9DED4] px-4 py-3 text-sm outline-none transition focus:border-[#17472D] focus:ring-2 focus:ring-[#17472D]/10"

                />

                <p className="text-xs text-[#718071]">

                  Enter 10â€“3000 characters. Your message will appear in the recipient's inquiry inbox.

                </p>

                <div className="flex flex-col-reverse gap-3 sm:flex-row sm:justify-end">

                  <button

                    type="button"

                    onClick={() => setContactListing(null)}

                    disabled={inquirySending}

                    className="rounded-lg border border-[#D9DED4] px-4 py-3 text-sm font-bold text-[#526755] hover:bg-[#F7F8F4] disabled:opacity-50"

                  >

                    Cancel

                  </button>

                  <button

                    type="submit"

                    disabled={inquirySending || inquiryMessage.trim().length < 10}

                    className="inline-flex items-center justify-center gap-2 rounded-lg bg-[#17472D] px-5 py-3 text-sm font-bold text-white hover:bg-[#235B3A] disabled:cursor-not-allowed disabled:opacity-60"

                  >

                    <Send size={16} />

                    {inquirySending ? "Sending..." : "Send inquiry"}

                  </button>

                </div>

              </form>

            )}

            {inquiryFeedbackType === "success" && (

              <button

                type="button"

                onClick={() => setContactListing(null)}

                className="mt-5 w-full rounded-lg bg-[#17472D] px-5 py-3 text-sm font-bold text-white hover:bg-[#235B3A]"

              >

                Close

              </button>

            )}

          </section>

        </div>

      )}

      <section className="border-t border-[#E7E4D9] bg-white">

        <div className="mx-auto grid max-w-7xl gap-5 px-4 py-10 sm:px-6 md:grid-cols-3 lg:px-8">

          <div className="flex gap-3">

            <ShieldCheck

              className="shrink-0 text-[#17472D]"

              size={23}

            />

            <div>

              <h3 className="font-bold">Build trust</h3>

              <p className="mt-1 text-sm leading-6 text-[#718071]">

                Review profiles, discuss terms and verify qualifications

                before making hiring decisions.

              </p>

            </div>

          </div>

          <div className="flex gap-3">

            <CalendarClock

              className="shrink-0 text-[#17472D]"

              size={23}

            />

            <div>

              <h3 className="font-bold">Agree on availability</h3>

              <p className="mt-1 text-sm leading-6 text-[#718071]">

                Confirm work dates, job requirements, rates and

                availability before arranging a booking.

              </p>

            </div>

          </div>

          <div className="flex gap-3">

            <Wrench

              className="shrink-0 text-[#17472D]"

              size={23}

            />

            <div>

              <h3 className="font-bold">Support the whole farm</h3>

              <p className="mt-1 text-sm leading-6 text-[#718071]">

                Connect with suppliers, labourers, equipment providers

                and agricultural professionals.

              </p>

            </div>

          </div>

        </div>

      </section>

      <footer className="bg-[#103720] px-4 py-8 text-white">

        <div className="mx-auto flex max-w-7xl flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">

          <div>

            <a href="/" className="text-lg font-extrabold">

              AgriVanta

            </a>

            <p className="mt-1 text-sm text-white/70">

              Powering Every Link in Agriculture.

            </p>

          </div>

          <a

            href="/"

            className="inline-flex items-center gap-2 text-sm font-semibold text-white/80 hover:text-white"

          >

            Back to homepage

            <ArrowRight size={16} />

          </a>

        </div>

      </footer>

    </main>

  );

}
